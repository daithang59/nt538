\section{Challenge 5: Tìm kiếm trong mảng số nguyên lớn}
\label{sec:challenge5}

\begin{requirementbox}{Yêu cầu bài toán}
Input gồm khóa cần tìm $K$ và một mảng số nguyên $A$ có thể chứa tối đa $10^6$ phần tử. Chương trình cần trả về chỉ số nhỏ nhất $i$ sao cho $A[i] = K$; nếu không tồn tại, kết quả là $-1$. Đề bài yêu cầu tự chia mảng và không sử dụng hàm tìm kiếm dựng sẵn.
\end{requirementbox}

\subsection{Hiện trạng mã nguồn}
Tại thời điểm lập báo cáo, \codepath{challenge/challenge5.py} chưa có nội dung hiện thực. Vì vậy phần này được trình bày như một thiết kế đề xuất dựa trên yêu cầu của đề bài. Khi nhóm bổ sung mã nguồn cuối cùng cho challenge này, cần cập nhật lại mô tả hiện thực để phản ánh chính xác phiên bản đã nộp.

\subsection{Thiết kế đề xuất}
Chiến lược song song hóa phù hợp là chia mảng $A$ thành các đoạn chỉ số liên tiếp. Mỗi worker quét tuyến tính đoạn được giao và trả về vị trí đầu tiên tìm thấy $K$ trong đoạn đó. Tiến trình chính lấy chỉ số nhỏ nhất trong các kết quả hợp lệ.

Cách tiếp cận này bảo đảm tính đúng đắn vì mọi đoạn tạo thành một phân hoạch của mảng ban đầu. Nếu một worker trả về chỉ số trong đoạn của nó, đó là vị trí đầu tiên của $K$ trong đoạn đó. Lấy giá trị nhỏ nhất trên toàn bộ các đoạn sẽ cho ra vị trí đầu tiên của $K$ trong toàn mảng.

\subsection{Thiết kế thuật toán song song}
\begin{lstlisting}[caption={Pseudocode đề xuất cho Challenge 5}]
MAIN(path):
    K, A = parse_input(path)
    ranges = split_index_ranges(len(A))

    candidates = pool.map(search_range, ranges)
    valid = [idx for idx in candidates if idx != -1]
    if valid is empty:
        return -1
    return min(valid)

search_range(start, end):
    for i in range(start, end):
        if A[i] == K:
            return i
    return -1
\end{lstlisting}

\subsection{Dữ liệu, phụ thuộc và chi phí}
\begin{itemize}
  \item \textbf{Phân phối dữ liệu}: mảng $A$ có thể được lưu trong biến toàn cục; worker chỉ nhận khoảng chỉ số cần quét.
  \item \textbf{Phụ thuộc dữ liệu}: các đoạn tìm kiếm độc lập; bước reduce chỉ chọn chỉ số nhỏ nhất.
  \item \textbf{Cân bằng tải}: nếu không có dừng sớm toàn cục, mỗi worker quét số phần tử gần bằng nhau.
  \item \textbf{Chi phí giao tiếp}: thấp vì mỗi worker chỉ trả về một số nguyên.
\end{itemize}
